1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "ghcp-usage-dashboard.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        You can watch its status with: kubectl get svc --namespace {{ .Release.Namespace }} -w {{ include "ghcp-usage-dashboard.fullname" . }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "ghcp-usage-dashboard.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "ghcp-usage-dashboard.fullname" . }} 8080:{{ .Values.service.port }}
  echo "Visit http://127.0.0.1:8080 to use the dashboard"
{{- end }}

IMPORTANT NOTES:
  ⚠️  This application uses SQLite for storage. Only ONE replica can run at a time.
      Do not scale beyond 1 replica when persistence is enabled.

  📦 Data is persisted to a PVC mounted at /app/data.
      Back up the SQLite database regularly for disaster recovery.

  🔐 Ensure GitHub tokens are configured via Helm values or an existing secret.
      Set .Values.existingSecret to use an external secret manager (Sealed Secrets, ESO, etc.).

  🌐 Ingress is {{ if .Values.ingress.enabled }}enabled{{ else }}disabled{{ end }}.
      {{ if not .Values.ingress.enabled }}Enable it in values.yaml and configure TLS + authentication before exposing publicly.{{ end }}
